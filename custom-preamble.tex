% --- Caption Styling: Centered ---
\captionsetup{justification=centering, font=singlespacing}
\captionsetup[table]{position=top, skip=6pt}
\captionsetup[figure]{position=bottom, skip=6pt}

% --- Page Numbering: Top-right, but bottom-center on chapter start ---
\fancypagestyle{plain}{
  \fancyhf{}
  \fancyfoot[C]{\thepage} % Chapter title page: centered at the bottom
  \renewcommand{\headrulewidth}{0pt}
}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[R]{\thepage}   % Normal page: top right
\renewcommand{\headrulewidth}{0pt}

% --- Chapter Title: Centered, Bold, BAB X ---
\titleformat{\chapter}[display]
  {\normalfont\bfseries\centering}
  {\MakeUppercase{\chaptertitlename\ \Roman{chapter}}}
  {0.5ex}
  {\MakeUppercase}
\titlespacing*{\chapter}{0pt}{-20pt}{20pt}

% --- Sub-headings (rata kiri, kapitalisasi awal kata) ---
\titleformat{\section}
  {\normalfont\bfseries}{\Roman{chapter}.\arabic{section}.}{0.5em}{}
\titleformat{\subsection}
  {\normalfont\bfseries}{\Roman{chapter}.\arabic{section}.\arabic{subsection}.}{0.5em}{}

% --- Numbering formats for equations, tables, figures (Bab.Nomor) ---
\renewcommand{\theequation}{\arabic{chapter}.\arabic{equation}}
\renewcommand{\thefigure}{\arabic{chapter}.\arabic{figure}}
\renewcommand{\thetable}{\arabic{chapter}.\arabic{table}}
